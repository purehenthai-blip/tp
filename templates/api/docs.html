Quick Start Example

# 1. Get JWT token
curl -X POST /api/v1/auth/token/ \
  -H "Content-Type: application/json" \
  -d '{"username": "youruser", "password": "yourpass"}'

# Response: {"access": "eyJ...", "refresh": "eyJ..."}

# 2. Use token to browse listings
curl /api/v1/listings/ \
  -H "Authorization: Bearer eyJ..."

# 3. Place an auction bid
curl -X POST /api/v1/auctions/{id}/bid/ \
  -H "Authorization: Bearer eyJ..." \
  -H "Content-Type: application/json" \
  -d '{"bid_amount": "0.05"}'
